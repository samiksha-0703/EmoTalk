# EmoTalk Backend — Python dependencies
# Target: Python 3.12 (developed/validated on CPython 3.12.4)
#
# This file covers the FastAPI backend and its runtime inference stack.
# The runtime path is: app.main -> app.api.routes -> app.core.hf_model_loader
#                                                 -> app.services.{database,gemini}
#
# Inference uses PyTorch + HuggingFace Transformers (superb/wav2vec2-base-superb-er),
# NOT TensorFlow. The TensorFlow training pipeline lives in ai_core/ and has its
# own requirements file (see ai_core/requirements.txt).

# --- Web framework / API server ---
fastapi>=0.110,<1.0          # FastAPI, CORSMiddleware, APIRouter, UploadFile, Depends (app.main, app.api.routes)
uvicorn[standard]>=0.27,<1.0 # ASGI server used to run the app: `uvicorn app.main:app`
python-multipart>=0.0.9      # Required by FastAPI to parse multipart/form-data uploads (UploadFile/File in /predict)
pydantic[email]>=2.5,<3.0    # BaseModel + EmailStr response/request models (app.api.routes); [email] pulls email-validator
python-dotenv>=1.0,<2.0      # load_dotenv() reads backend/.env into environment (app.core.config)

# --- Authentication ---

torch>=2.2,<3.0             # Model tensors + inference (app.core.hf_model_loader)

# --- Google Gemini insights ---
google-generativeai>=0.5,<1.0  # genai.configure / GenerativeModel (app.services.gemini)
